const MAX_IMAGE_BYTES = 5 * 1024 * 1024; const PASSWORD_ITERATIONS = 120000; const PASSWORD_KEY = "admin-password-v1"; function jsonResponse(data, status, origin) { return new Response(JSON.stringify(data), { status, headers: { "Access-Control-Allow-Origin": origin, "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization", "Access-Control-Max-Age": "86400", "Cache-Control": "no-store", "Content-Type": "application/json;
